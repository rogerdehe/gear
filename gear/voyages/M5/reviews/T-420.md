use gear_store::{Store, testing};
use gear_model::{SeatRole, SeatState};
use gear_herdr::fake::FakeHerdr;
fn main() {

        let wt = root.join(if alias {"with-alias"} else {"review-key-only"});

            testing::seed_project(tx, "p", "sess", 1000)?;

            tx.set_meta("engine.review_path.p:review-1", wt.to_str().unwrap())?;
            if alias { tx.set_meta("engine.worktree_path.p:review-1", wt.to_str().unwrap())?; }
